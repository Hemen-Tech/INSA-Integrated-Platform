def read_csv_bytes(data: bytes) -> pd.DataFrame:
    return pd.read_csv(io.BytesIO(data), low_memory=False)


def load_uploaded(uploaded_file):
    name = uploaded_file.name.lower()
    raw = uploaded_file.getvalue()

    if name.endswith(".csv"):
        return read_csv_bytes(raw)

    if name.endswith((".xlsx", ".xls")):
        return pd.read_excel(io.BytesIO(raw))

    if name.endswith(".json"):
        obj = json.loads(raw.decode("utf-8"))
        if isinstance(obj, dict):
            return pd.DataFrame([
                {
                    "audit_key": key,
                    "audit_value": (
                        json.dumps(value)
                        if isinstance(value, (dict, list))
                        else value
                    ),
                }
                for key, value in obj.items()
            ])
        return pd.DataFrame(obj)

    if name.endswith((".gpkg", ".geojson", ".shp", ".zip")):
        if gpd is None:
            raise RuntimeError(
                "GeoPandas is unavailable. Install it in the active environment."
            )

        suffix = Path(name).suffix
        with tempfile.NamedTemporaryFile(delete=False, suffix=suffix) as tmp:
            tmp.write(raw)
            temp_path = tmp.name

        try:
            gdf = gpd.read_file(temp_path)
            if gdf.crs is not None:
                gdf = gdf.to_crs("EPSG:4326")
            return gdf
        finally:
            try:
                os.unlink(temp_path)
            except OSError:
                pass

    raise ValueError(f"Unsupported file type: {uploaded_file.name}")


def auto_load_local_files():
    """Load existing PNIS artifacts without modifying source files."""
    result = {}

    if not PROCESSED_DIR.exists():
        result["data directory load error"] = (
            f"Processed directory not found: {PROCESSED_DIR}"
        )
        return result

    for label, filename in EXPECTED_FILES.items():
        path = PROCESSED_DIR / filename

        if not path.is_file():
            result[f"{label} load error"] = f"File not found: {path}"
            continue

        try:
            suffix = path.suffix.lower()

            if suffix == ".csv":
                result[label] = pd.read_csv(path, low_memory=False)

            elif suffix in (".xlsx", ".xls"):
                result[label] = pd.read_excel(path)

            elif suffix == ".json":
                result[label] = json.loads(
                    path.read_text(encoding="utf-8")
                )

            elif suffix == ".gpkg":
                if gpd is None:
                    result[f"{label} load error"] = (
                        "GeoPandas is not installed in the active environment."
                    )
                    continue

                # Explicitly load the known Phase 2C V2 road layer.
                if filename == ROAD_FILENAME:
                    gdf = gpd.read_file(path, layer=ROAD_LAYER)
                else:
                    gdf = gpd.read_file(path)

                if gdf.empty:
                    result[f"{label} load error"] = (
                        f"The GeoPackage layer is empty: {path.name}"
                    )
                    continue

                if gdf.crs is None:
                    result[f"{label} load error"] = (
                        f"CRS missing from {path.name}; refusing to guess coordinates."
                    )
                    continue

                # Web maps expect longitude/latitude.
                result[label] = gdf.to_crs("EPSG:4326")

        except Exception as exc:
            result[f"{label} load error"] = (
                f"{type(exc).__name__}: {exc}"
            )

    return result


def find_col(df, patterns):
    if df is None or df.empty:
        return None

    cols = list(df.columns)
    normalized = {
        str(col).lower().replace(" ", "_"): col
        for col in cols
    }

    for pattern in patterns:
        for name, original in normalized.items():
            if pattern in name:
                return original

    return None


def display_count(value):
    return f"{value:,}" if value is not None else "â€”"


def dataframe_for_map(data):
    if not isinstance(data, pd.DataFrame) or data.empty:
        return None

    if hasattr(data, "geometry"):
        return data

    lat = find_col(data, ["latitude", "lat"])
    lon = find_col(data, ["longitude", "lon", "long"])

    if lat and lon:
        df = data.copy()
        df["_lat"] = pd.to_numeric(df[lat], errors="coerce")
        df["_lon"] = pd.to_numeric(df[lon], errors="coerce")

        # Reject invalid geographic coordinates.
        df = df.dropna(subset=["_lat", "_lon"])
        df = df[
            df["_lat"].between(-90, 90)
            & df["_lon"].between(-180, 180)
        ]
        return df

    return None


def to_geojson(gdf):
    if gdf is None or gdf.empty or not hasattr(gdf, "geometry"):
        return None

    try:
        return json.loads(gdf.to_json())
    except Exception:
        return None

    )

    # ---------------- Road geometry ----------------

    if show_roads and roads is not None and hasattr(roads, "geometry") and not roads.empty:
        geojson = to_geojson(roads)

        if geojson:
            map_layers.append(
                pdk.Layer(
                    "GeoJsonLayer",
                    data=geojson,
                    stroked=True,
                    filled=False,
                    get_line_color=[69, 177, 118, 190],
                    get_line_width=1.5,
                    line_width_min_pixels=1,
                    pickable=True,
                    auto_highlight=True,
                )
            )

            map_data_present = True

            try:
                bounds = roads.total_bounds

                if len(bounds) == 4 and all(pd.notna(x) for x in bounds):
                    min_lon, min_lat, max_lon, max_lat = bounds

                    center_lat = (min_lat + max_lat) / 2
                    center_lon = (min_lon + max_lon) / 2

                    span = max(max_lon - min_lon, max_lat - min_lat)

                    if span > 0:
                        import math
                        zoom = max(5, min(13, math.log2(360 / span) - 1))
                    else:
                        zoom = 10

                    view_state = pdk.ViewState(
                        latitude=float(center_lat),
                        longitude=float(center_lon),
                        zoom=float(zoom),
                        pitch=0,
                        bearing=0,
                    )

            except Exception:
                pass

    # ---------------- Candidate locations ----------------

    candidate_map = (
        dataframe_for_map(filtered_worklist)
        if show_candidates
        else None
    )

    if candidate_map is not None and not candidate_map.empty:
        if hasattr(candidate_map, "geometry"):
            gj = to_geojson(candidate_map)

            if gj:
                map_layers.append(
                    pdk.Layer(
                        "GeoJsonLayer",
                        data=gj,
                        stroked=True,
                        filled=True,
                        get_fill_color=[183, 228, 91, 150],
                        get_line_color=[220, 245, 175, 220],
                        get_line_width=2,
                        point_radius_min_pixels=5,
                        get_point_radius=90,
                        pickable=True,
                        auto_highlight=True,
                    )
                )
                map_data_present = True

        elif "_lat" in candidate_map.columns and "_lon" in candidate_map.columns:
            map_layers.append(
                pdk.Layer(
                    "ScatterplotLayer",
                    data=candidate_map,
                    get_position=["_lon", "_lat"],
                    get_radius=100,
                    radius_min_pixels=4,
                    get_fill_color=[183, 228, 91, 210],
                    get_line_color=[238, 245, 220, 230],
                    line_width_min_pixels=1,
                    stroked=True,
                    pickable=True,
                )
            )
            map_data_present = True

    # ---------------- Graph nodes ----------------

    if show_nodes and nodes is not None and not nodes.empty:
        lat_col = find_col(nodes, ["latitude", "lat"])
        lon_col = find_col(nodes, ["longitude", "lon", "long"])

        if lat_col and lon_col:
            node_map = nodes.copy()
            node_map["_lat"] = pd.to_numeric(node_map[lat_col], errors="coerce")
            node_map["_lon"] = pd.to_numeric(node_map[lon_col], errors="coerce")

            node_map = node_map.dropna(subset=["_lat", "_lon"])
            node_map = node_map[
                node_map["_lat"].between(-90, 90)
                & node_map["_lon"].between(-180, 180)
            ]

            if not node_map.empty:
                map_layers.append(
                    pdk.Layer(
                        "ScatterplotLayer",
                        data=node_map,
                        get_position=["_lon", "_lat"],
                        get_radius=60,
                        radius_min_pixels=3,
                        get_fill_color=[97, 214, 209, 200],
                        pickable=True,
                    )
                )
                map_data_present = True

    # ---------------- Render map ----------------

    deck = pdk.Deck(
        layers=map_layers,
        initial_view_state=view_state,
        map_style="https://basemaps.cartocdn.com/gl/dark-matter-gl-style/style.json",
        tooltip={"text": "{name}"},
    )

    st.pydeck_chart(
        deck,
        use_container_width=True,
        height=440,
        key="pnis_map",
    )

    if not map_data_present:
        st.warning(
            "No valid road geometry or candidate/node coordinates are currently "
            "available for the enabled map layers. Open Map diagnostics below."
        )

    st.caption(
        "Map features show source geometry or coordinates. "
        "Geometric connectivity does not prove passability or route approval."
    )

    # ---------------- Map diagnostics ----------------

    with st.expander("Map diagnostics"):
        st.write("Application directory:", str(APP_DIR))
        st.write("Project root:", str(PROJECT_ROOT))
        st.write("Processed data directory:", str(PROCESSED_DIR))
        st.write("GeoPandas available:", gpd is not None)
        st.write("Road dataset loaded:", roads is not None)

        if roads is not None:
            st.write("Road features:", f"{len(roads):,}")
            st.write("Road CRS:", str(roads.crs))
            st.write(
                "Geometry types:",
                roads.geometry.geom_type.value_counts().to_dict(),
            )
            st.write("Bounds (longitude/latitude):", roads.total_bounds.tolist())

        if candidate_map is not None:
            st.write("Candidate map rows:", f"{len(candidate_map):,}")
            st.write(
                "Candidate coordinate fields:",
                [
                    column for column in ("_lat", "_lon")
                    if column in candidate_map.columns
                ],
            )

        if load_errors:
            st.error("One or more local data sources failed to load.")
            st.json(load_errors)

        elif roads is None:
            st.warning(
                "Road geometry is not loaded. Check the GeoPackage path, "
                "layer name, and GeoPandas installation."
            )

        elif not map_data_present:
            st.warning(
                "The road data loaded, but no enabled layer produced map features. "
                "Check layer toggles, geometry, and coordinates."
            )

        else:
            st.success("At least one map layer contains renderable source features.")

    # ---------------- Candidate worklist ----------------

    st.markdown("<div style='height:10px'></div>", unsafe_allow_html=True)

    st.markdown(
        '<div class="panel-title">Candidate worklist</div>'
        '<div class="panel-sub">Filtered records Â· review candidates are not automatically routing-approved</div>',
        unsafe_allow_html=True,
    )

    if not filtered_worklist.empty:
        exp_col, rows_col = st.columns([1, 2])

        with exp_col:
            st.download_button(
                "â†“ Export filtered CSV",
                filtered_worklist.to_csv(index=False).encode("utf-8-sig"),
                file_name="pnis_filtered_worklist.csv",
                mime="text/csv",
                use_container_width=True,
            )

        with rows_col:
            st.caption(f"{len(filtered_worklist):,} records in current view")

        st.dataframe(
            filtered_worklist,
            use_container_width=True,
            hide_index=True,
            height=230,
        )

        with st.expander("Inspect selected record"):
            row_idx = st.selectbox(
                "Choose record row",
                options=list(range(len(filtered_worklist))),
                format_func=lambda i: (
                    f"Row {i + 1} Â· {filtered_worklist.iloc[i, 0]}"
                ),
            )
            st.json(
                filtered_worklist.iloc[row_idx].to_dict(),
                expanded=True,
            )

    else:
        st.info(
            "Candidate worklist is unavailable or empty. "
            "Check the source file and Map diagnostics."
        )
